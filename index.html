관제 연동 설정 & 구글 시트 가이드

※ N/A 처리 시 0명 데이터 자동 기입(POST)을 지원하려면 웹앱에 doPost 핸들러가 구현되어 있어야 합니다.

Google Apps Script 연동 코드 예시 (doGet + doPost) 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var params = e.parameter;
  if (!params.dept) {
    try { params = JSON.parse(e.postData.contents); } catch(err){}
  }
  var timestamp = params.timestamp || new Date();
  sheet.appendRow([
    timestamp,
    params.dept,
    params.submitter || "관리자(NA)",
    Number(params.eTotal) || 0,
    Number(params.eDone) || 0,
    Number(params.gTotal) || 0,
    Number(params.gDone) || 0,
    params.note || "[N/A 열외]"
  ]);
  return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT);
}
