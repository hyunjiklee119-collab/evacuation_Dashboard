관제 연동 설정 & 구글 시트 가이드

※ 관리자 수기 점호 등록 및 N/A 0명 기입(POST) 지원을 위해 doPost 핸들러가 배포되어 있어야 합니다.

Google Apps Script 연동 코드 예시 (doGet + doPost) 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var params = e.parameter;
  if (!params || !params.dept) {
    try { params = JSON.parse(e.postData.contents); } catch(err){}
  }
  var timestamp = params.timestamp || new Date();
  sheet.appendRow([
    timestamp,
    params.dept,
    params.submitter || "관리자 대리입력",
    Number(params.eTotal || params.empTotal) || 0,
    Number(params.eDone || params.empDone) || 0,
    Number(params.gTotal || params.guestTotal) || 0,
    Number(params.gDone || params.guestDone) || 0,
    params.note || "-"
  ]);
  return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT);
}
